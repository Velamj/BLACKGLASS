cmake_minimum_required(VERSION 3.20)
project(BlackglassCore LANGUAGES CXX)
# This target is a simulation library and tests, not an Unreal game.
add_library(blackglass_core STATIC Source/BlackglassCore/core.cpp)
target_include_directories(blackglass_core PUBLIC Source/BlackglassCore/include)
target_compile_features(blackglass_core PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(blackglass_core PRIVATE /W4 /WX /permissive- /EHsc)
else()
  target_compile_options(blackglass_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_executable(blackglass_core_tests Tests/core_tests.cpp)
target_link_libraries(blackglass_core_tests PRIVATE blackglass_core)
if(MSVC)
  target_compile_options(blackglass_core_tests PRIVATE /W4 /WX /permissive- /EHsc)
else()
  target_compile_options(blackglass_core_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME blackglass_core_contracts COMMAND blackglass_core_tests)
set_tests_properties(blackglass_core_contracts PROPERTIES TIMEOUT 30 WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
